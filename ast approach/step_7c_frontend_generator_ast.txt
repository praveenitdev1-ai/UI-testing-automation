"#!/usr/bin/env python3
"""
Frontend Generator AST - Phase 2 Task 3
========================================
Generates transformed TSX code using AST-based analysis and wiring.

This script:
1. Runs the AST extractor (step_7a) to get metadata
2. Runs the wiring analyzer (step_7b) to get wiring config
3. Calls the Node.js AST transformer to apply transformations
4. Outputs the transformed TSX file

Usage:
    python step_7c_frontend_generator_ast.py -i Inputs/ui_react_component.tsx -o Generated/frontend/src/components/NSIManagement.tsx
"""

import subprocess
import json
import argparse
import logging
import shutil
import sys
from pathlib import Path
from datetime import datetime
from typing import Dict, Any, Optional, List

# Add Templates to path for ui_capabilities import
sys.path.insert(0, str(Path(__file__).parent / 'Templates'))

# Import UI Capability Renderer for config-driven code generation
try:
    from ui_capabilities.pattern_matcher import UIPatternMatcher
    from ui_capabilities.renderer import UICapabilityRenderer, RenderedCapability
    UI_CAPABILITIES_AVAILABLE = True
except ImportError:
    UI_CAPABILITIES_AVAILABLE = False

# Setup logging
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s'
)
logger = logging.getLogger(__name__)


class FrontendGeneratorAST:
    """
    Orchestrates the AST-based frontend generation pipeline.

    Pipeline:
    1. AST Extraction (tsx_ast_extractor.js)
    2. Wiring Analysis (step_7b_frontend_wiring_ast.py)
    3. AST Transformation (tsx_ast_transformer.js)
    """

    def __init__(self, tsx_input: str, output_path: str):
        self.tsx_input = Path(tsx_input)
        self.output_path = Path(output_path)
        self.script_dir = Path(__file__).parent

        # Tool paths
        self.ast_tools_dir = self.script_dir / 'ast_tools'
        self.config_output_dir = self.script_dir / 'config_output'

        # Intermediate files
        self.ast_metadata_path = self.config_output_dir / 'tsx_ast_metadata.json'
        self.wiring_config_path = self.config_output_dir / 'wired_ui_ast.json'

        # Validate input
        if not self.tsx_input.exists():
            raise FileNotFoundError(f"TSX input file not found: {self.tsx_input}")

        # Ensure output directory exists
        self.output_path.parent.mkdir(parents=True, exist_ok=True)

        # CONFIG-DRIVEN: Load application config for entity information
        self.app_config = self._load_application_config()
        self.primary_entity = self._get_primary_entity()
        self.app_metadata = self.app_config.get('metadata', {})
        logger.info(f"Config-driven mode: primary_entity={self.primary_entity}")

    def _load_application_config(self) -> Dict[str, Any]:
        """Load application_config.json for entity and app metadata."""
        config_path = self.config_output_dir / 'application_config.json'
        if config_path.exists():
            with open(config_path) as f:
                return json.load(f)
        logger.warning("application_config.json not found - using defaults")
        return {}

    def _get_primary_entity(self) -> str:
        """
        Determine the primary entity from config.

        The primary entity is the main entity that drives the UI.
        Determined by:
        1. Explicit primary_entity in config
        2. First entity in database_schema.tables with most columns
        3. Fallback to first entity in schema
        """
        # Check for explicit primary_entity
        if 'primary_entity' in self.app_config:
            return self.app_config['primary_entity']

        # Get from database_schema
        db_schema = self.app_config.get('database_schema', {})
        tables = db_schema.get('tables', [])

        if not tables:
            logger.warning("No tables found in config - using default entity")
            return 'items'  # Generic fallback

        # Find table with most columns (likely the main entity)
        primary_table = max(tables, key=lambda t: len(t.get('columns', [])))
        return primary_table.get('name', 'items')

    def _get_entity_fields(self, entity_name: str) -> List[Dict[str, Any]]:
        """Get field definitions for an entity from config."""
        db_schema = self.app_config.get('database_schema', {})
        for table in db_schema.get('tables', []):
            if table.get('name') == entity_name:
                return table.get('columns', [])
        return []

    def _get_primary_key(self, entity_name: str) -> str:
        """Get primary key field name for an entity."""
        db_schema = self.app_config.get('database_schema', {})
        for table in db_schema.get('tables', []):
            if table.get('name') == entity_name:
                pks = table.get('primary_keys', [])
                if pks:
                    return pks[0]
                # Fallback: find column with primary_key=True
                for col in table.get('columns', []):
                    if col.get('primary_key'):
                        return col.get('name')
        return 'id'  # Generic fallback

    def run_pipeline(self, skip_extraction: bool = False, skip_wiring: bool = False) -> Dict[str, Any]:
        """
        Run the complete frontend generation pipeline.

        Args:
            skip_extraction: If True, use existing AST metadata
            skip_wiring: If True, use existing wiring config

        Returns:
            Pipeline execution summary
        """
        summary = {
            'started_at': datetime.now().isoformat(),
            'steps': [],
            'success': False
        }

        try:
            # Step 1: AST Extraction
            if not skip_extraction:
                logger.info("Step 1: Running AST extraction...")
                extraction_result = self._run_ast_extraction()
                summary['steps'].append({
                    'name': 'ast_extraction',
                    'success': extraction_result['success'],
                    'output': str(self.ast_metadata_path)
                })
                if not extraction_result['success']:
                    raise RuntimeError(f"AST extraction failed: {extraction_result.get('error')}")
            else:
                logger.info("Step 1: Skipping AST extraction (using existing)")
                summary['steps'].append({
                    'name': 'ast_extraction',
                    'success': True,
                    'skipped': True
                })

            # Step 2: Wiring Analysis
            if not skip_wiring:
                logger.info("Step 2: Running wiring analysis...")
                wiring_result = self._run_wiring_analysis()
                summary['steps'].append({
                    'name': 'wiring_analysis',
                    'success': wiring_result['success'],
                    'output': str(self.wiring_config_path)
                })
                if not wiring_result['success']:
                    raise RuntimeError(f"Wiring analysis failed: {wiring_result.get('error')}")
            else:
                logger.info("Step 2: Skipping wiring analysis (using existing)")
                summary['steps'].append({
                    'name': 'wiring_analysis',
                    'success': True,
                    'skipped': True
                })

            # Step 3: AST Transformation
            logger.info("Step 3: Running AST transformation...")
            transform_result = self._run_ast_transformation()
            summary['steps'].append({
                'name': 'ast_transformation',
                'success': transform_result['success'],
                'output': str(self.output_path),
                'modifications': transform_result.get('modifications', {})
            })

            if not transform_result['success']:
                raise RuntimeError(f"AST transformation failed: {transform_result.get('error')}")

            # Step 4: Post-process App.tsx (add TypeScript types, fix imports)
            logger.info("Step 4: Post-processing App.tsx...")
            postprocess_result = self._postprocess_app_tsx()
            summary['steps'].append({
                'name': 'postprocess_app_tsx',
                'success': postprocess_result['success'],
                'fixes': postprocess_result.get('fixes', [])
            })

            # Step 5: Clean up orphan components (modals from LLM version, etc.)
            logger.info("Step 5: Cleaning up orphan components...")
            cleanup_result = self._cleanup_orphan_components()
            summary['steps'].append({
                'name': 'cleanup_orphans',
                'success': cleanup_result['success'],
                'removed': cleanup_result.get('removed', [])
            })

            # Step 6: Generate supporting files (index.tsx, etc.)
            logger.info("Step 6: Generating supporting files...")
            support_result = self._generate_supporting_files()
            summary['steps'].append({
                'name': 'supporting_files',
                'success': support_result['success'],
                'files': support_result.get('files', [])
            })

            summary['success'] = True
            summary['output_file'] = str(self.output_path)

        except Exception as e:
            logger.error(f"Pipeline failed: {e}")
            summary['error'] = str(e)

        summary['completed_at'] = datetime.now().isoformat()
        return summary

    def _run_ast_extraction(self) -> Dict[str, Any]:
        """Run the Node.js AST extractor."""
        try:
            # Ensure dependencies are installed
            self._ensure_node_dependencies()

            # Run the extractor
            extractor_script = self.ast_tools_dir / 'tsx_ast_extractor.js'
            result = subprocess.run(
                ['node', str(extractor_script), str(self.tsx_input)],
                capture_output=True,
                text=True,
                cwd=str(self.script_dir)
            )

            if result.returncode != 0:
                return {
                    'success': False,
                    'error': result.stderr
                }

            # Parse and save metadata
            metadata = json.loads(result.stdout)
            with open(self.ast_metadata_path, 'w') as f:
                json.dump(metadata, f, indent=2)

            logger.info(f"AST metadata saved to {self.ast_metadata_path}")
            return {'success': True, 'metadata': metadata}

        except Exception as e:
            return {'success': False, 'error': str(e)}

    def _run_wiring_analysis(self) -> Dict[str, Any]:
        """Run the wiring analysis script."""
        try:
            # Import and run the wiring script
            wiring_script = self.script_dir / 'step_7b_frontend_wiring_ast.py'

            result = subprocess.run(
                [
                    'python3', str(wiring_script),
                    '-a', str(self.config_output_dir / 'application_config.json'),  # -a is app-config
                    '-t', str(self.ast_metadata_path),  # -t is tsx-metadata
                    '-o', str(self.wiring_config_path)
                ],
                capture_output=True,
                text=True,
                cwd=str(self.script_dir)
            )

            if result.returncode != 0:
                return {
                    'success': False,
                    'error': result.stderr
                }

            logger.info(f"Wiring config saved to {self.wiring_config_path}")
            return {'success': True}

        except Exception as e:
            return {'success': False, 'error': str(e)}

    def _run_ast_transformation(self) -> Dict[str, Any]:
        """Run the Node.js AST transformer."""
        try:
            transformer_script = self.ast_tools_dir / 'tsx_ast_transformer.js'

            result = subprocess.run(
                [
                    'node', str(transformer_script),
                    str(self.tsx_input),
                    str(self.wiring_config_path),
                    str(self.output_path)
                ],
                capture_output=True,
                text=True,
                cwd=str(self.script_dir)
            )

            if result.returncode != 0:
                return {
                    'success': False,
                    'error': result.stderr
                }

            # Parse modification summary from stderr
            modifications = self._parse_modification_summary(result.stderr)

            logger.info(f"Transformed TSX saved to {self.output_path}")
            return {
                'success': True,
                'modifications': modifications
            }

        except Exception as e:
            return {'success': False, 'error': str(e)}

    def _ensure_node_dependencies(self):
        """Ensure Node.js dependencies are installed."""
        node_modules = self.ast_tools_dir / 'node_modules'
        if not node_modules.exists():
            logger.info("Installing Node.js dependencies...")
            result = subprocess.run(
                ['npm', 'install'],
                capture_output=True,
                text=True,
                cwd=str(self.ast_tools_dir)
            )
            if result.returncode != 0:
                logger.warning(f"npm install warning: {result.stderr}")

    def _parse_modification_summary(self, stderr: str) -> Dict[str, int]:
        """Parse modification counts from transformer stderr output."""
        modifications = {}

        lines = stderr.split('\n')
        for line in lines:
            if 'Imports added:' in line:
                modifications['imports'] = int(line.split(':')[1].strip())
            elif 'Hooks injected:' in line:
                modifications['hooks'] = int(line.split(':')[1].strip())
            elif 'Handlers wrapped:' in line:
                modifications['handlers'] = int(line.split(':')[1].strip())
            elif 'IDs injected:' in line:
                modifications['ids'] = int(line.split(':')[1].strip())

        return modifications

    def _postprocess_app_tsx(self) -> Dict[str, Any]:
        """
        Post-process the generated App.tsx to:
        - Add type annotations to function parameters
        - Wire useNsiItems hook for API integration (shared data between screens)
        - Fix any import path issues
        """
        import re

        try:
            fixes = []

            # Read the generated App.tsx
            with open(self.output_path, 'r') as f:
                content = f.read()

            original_content = content

            # Fix 1: Add types to common callback parameters
            # Pattern: (message, type) => or (message,type)=>
            type_fixes = [
                # showNotification(message, type)
                (r'const showNotification = \(message, type\)',
                 'const showNotification = (message: string, type: string)'),
                (r'const showNotification=\(message,type\)',
                 'const showNotification = (message: string, type: string)'),

                # showModal(content)
                (r'const showModal = \(content\)',
                 'const showModal = (content: string)'),
                (r'const showModal=\(content\)',
                 'const showModal = (content: string)'),

                # showScreen(screenId)
                (r'const showScreen = \(screenId\)',
                 'const showScreen = (screenId: string)'),
                (r'const showScreen=\(screenId\)',
                 'const showScreen = (screenId: string)'),

                # Generic event handlers
                (r'\(e\) =>', '(e: React.ChangeEvent<HTMLInputElement | HTMLSelectElement>) =>'),
                (r'\(event\) =>', '(event: React.ChangeEvent<HTMLInputElement | HTMLSelectElement>) =>'),

                # onClick handlers with event
                (r'onClick=\{\(e\) =>', 'onClick={(e: React.MouseEvent) =>'),
                (r'onClick=\{\(event\) =>', 'onClick={(event: React.MouseEvent) =>'),

                # onChange with specific target value access - be more careful
                (r'onChange=\{\(e\) => \{', 'onChange={(e: React.ChangeEvent<HTMLSelectElement>) => {'),

                # item, index in map
                (r'\.map\(\(item, index\) =>', '.map((item: any, index: number) =>'),
                (r'\.map\(\(item\) =>', '.map((item: any) =>'),
            ]

            for pattern, replacement in type_fixes:
                if re.search(pattern, content):
                    content = re.sub(pattern, replacement, content)
                    fixes.append(f"Applied type fix: {pattern[:40]}...")

            # Fix 2: Wire API hook for data sharing between screens
            # CONFIG-DRIVEN: Use primary_entity from application_config.json
            if UI_CAPABILITIES_AVAILABLE:
                logger.info(f"Using config-driven UI capability wiring for entity: {self.primary_entity}")
                content = self._wire_api_data_sharing_v2(content, fixes, entity_name=self.primary_entity)
            else:
                logger.info("Using legacy hardcoded wiring (UI capabilities not available)")
                content = self._wire_api_data_sharing(content, fixes)

            # Write back if changed
            if content != original_content:
                with open(self.output_path, 'w') as f:
                    f.write(content)
                logger.info(f"Post-processed App.tsx with {len(fixes)} fixes")
            else:
                logger.info("No post-processing needed for App.tsx")

            return {'success': True, 'fixes': fixes}

        except Exception as e:
            logger.error(f"Failed to post-process App.tsx: {e}")
            return {'success': False, 'error': str(e), 'fixes': []}

    def _wire_api_data_sharing(self, content: str, fixes: list) -> str:
        """
        DEPRECATED LEGACY FALLBACK: Wire the entity hook to provide shared data.

        WARNING: This method is deprecated and should NOT be used.
        The preferred path is _wire_api_data_sharing_v2 which uses config-driven templates.

        This method only runs if:
        1. UI_CAPABILITIES_AVAILABLE is False (templates not installed)
        2. Or _wire_api_data_sharing_v2 throws an exception

        TODO: Remove this method entirely once config-driven approach is stable.
        """
        import re

        logger.warning("Using LEGACY hardcoded wiring - this should be avoided!")
        logger.warning("Please ensure UI Capability templates are properly installed")

        # Get entity name from config for dynamic hook naming
        entity_name = self.app_config.get('primary_entity', 'items')
        entity_pascal = ''.join(word.title() for word in entity_name.split('_'))
        entity_camel = entity_pascal[0].lower() + entity_pascal[1:]

        # Step 1: Update the entity hook destructure - MINIMAL version
        # NOTE: Handlers should come from templates, not hardcoded here
        old_destructure_pattern = rf"const \{{ create{entity_pascal}, update{entity_pascal}, delete{entity_pascal} \}} = use{entity_pascal}\(\);"
        new_destructure = f"""const {{
    data: {entity_camel},
    loading: {entity_camel}Loading,
    error: {entity_camel}Error,
    fetch{entity_pascal}s,
    create{entity_pascal},
    update{entity_pascal},
    delete{entity_pascal},
    lookupByTracking,
    checkDuplicate{entity_pascal},
    create{entity_pascal}WithCheck,
    setData: set{entity_pascal}
  }} = use{entity_pascal}();

  // State for auto-populated form fields (AC1, AC4)
  const [formData, setFormData] = React.useState<any>({{}});
  // Dynamic items list is now derived from entity API data (see getFilteredItems)
  // NOTE: Handlers (handleScanLookup, addToList) should be generated from templates"""

        if re.search(old_destructure_pattern, content):
            content = re.sub(old_destructure_pattern, new_destructure, content)
            fixes.append(f"Wired use{entity_pascal} hook (LEGACY fallback - no handlers)")

        # Step 2: Remove the hardcoded worklistItems useState (it's now derived from API)
        hardcoded_useState_pattern = r"const \[worklistItems, setWorklistItems\] = useState\(\[\s*\{[\s\S]*?\}\s*\]\);"
        if re.search(hardcoded_useState_pattern, content):
            content = re.sub(hardcoded_useState_pattern, "// worklistItems is now derived from useNsiItems hook data above", content)
            fixes.append("Replaced hardcoded worklistItems with API-driven data")

        # NOTE: addToWorklist replacement removed - should come from worklist_add.ts.j2 template
        # NOTE: handleScanLookup removed - should come from lookup_handler.ts.j2 template

        # Step 3: Wire Scan button to call handleScanLookup (AC1, AC4)
        # This is safe - just wiring, not generating the handler
        scan_button_pattern = r"onClick=\{\(\) => showNotification\('Scanning\.\.\.', 'success'\)\}"
        scan_button_replacement = "onClick={handleScanLookup}"
        if re.search(scan_button_pattern, content):
            content = re.sub(scan_button_pattern, scan_button_replacement, content)
            fixes.append("Wired Scan button to handleScanLookup for AC1/AC4 auto-populate")

        return content

    def _wire_api_data_sharing_v2(self, content: str, fixes: list, entity_name: str = None) -> str:
        """
        CONFIG-DRIVEN VERSION: Wire API data sharing using UI Capability Templates.

        Instead of hardcoded strings, this method:
        1. Reads entity_validations.json for lookup/validation config
        2. Reads wired_ui_ast.json for field ID mappings
        3. Uses UI Capability Templates to generate code

        This is the scalable approach - new user stories only need config, not code changes.
        """
        # Get entity name from config if not provided
        if entity_name is None:
            entity_name = self.app_config.get('primary_entity', 'items')
        import re

        if not UI_CAPABILITIES_AVAILABLE:
            logger.warning("UI Capabilities not available, falling back to hardcoded wiring")
            return self._wire_api_data_sharing(content, fixes)

        try:
            # Initialize capability renderer with config paths
            entity_validations_path = str(self.config_output_dir / 'entity_validations.json')
            wired_ui_ast_path = str(self.wiring_config_path)

            renderer = UICapabilityRenderer(entity_validations_path, wired_ui_ast_path)
            pattern_matcher = UIPatternMatcher(entity_validations_path, wired_ui_ast_path)

            # Discover patterns for this entity
            patterns = pattern_matcher.discover_patterns(entity_name)
            logger.info(f"Discovered {len(patterns)} UI patterns for {entity_name}: {[p.pattern_type for p in patterns]}")

            # Render all capabilities
            rendered_capabilities = renderer.render_entity(entity_name)

            # Group by injection point
            by_injection = {}
            for cap in rendered_capabilities:
                point = cap.injection_point
                if point not in by_injection:
                    by_injection[point] = []
                by_injection[point].append(cap)

            # Build the hook destructure dynamically from config
            entity_service = pattern_matcher.get_entity_service(entity_name)
            entity_pascal = entity_name.replace('_', ' ').title().replace(' ', '')

            # Check what capabilities are needed
            has_lookup = any(p.pattern_type == 'SCAN_LOOKUP' for p in patterns)
            has_duplicate_check = any(p.pattern_type == 'DUPLICATE_CHECK' for p in patterns)

            hook_functions = [
                f'data: {entity_name.replace("_", "")}',
                f'loading: {entity_name.replace("_", "")}Loading',
                f'error: {entity_name.replace("_", "")}Error',
                f'fetch{entity_pascal}s',
                f'create{entity_pascal}',
                f'update{entity_pascal}',
                f'delete{entity_pascal}',
            ]

            if has_lookup:
                lookup_entity = pattern_matcher.get_lookup_config(entity_name)
                if lookup_entity:
                    lookup_entity_name = lookup_entity.get('lookup_entity', 'source_data')
                    lookup_pascal = lookup_entity_name.replace('_', ' ').title().replace(' ', '')
                    hook_functions.append(f'lookup{lookup_pascal}ByTracking')

            if has_duplicate_check:
                hook_functions.append(f'checkDuplicate{entity_pascal}')
                hook_functions.append(f'create{entity_pascal}WithCheck')

            hook_functions.append(f'setData: set{entity_pascal}')

            # Build the new destructure statement
            hook_destructure = f"""const {{
    {','.join(chr(10) + '    ' + f for f in hook_functions)}
  }} = use{entity_pascal}();

  // State for auto-populated form fields (from config-driven lookup)
  const [formData, setFormData] = React.useState<any>({{}});"""

            # Build handlers from rendered capabilities
            # Indent each line by 2 spaces to be inside the component
            # NOTE: Exclude worklist_add since we handle addToWorklist replacement separately
            handlers_code = ""
            for cap in by_injection.get('handlers', []):
                # Skip worklist_add - it's replaced separately below using render_worklist_add()
                if cap.capability_name == 'worklist_add':
                    continue
                # Skip duplicate_checker - worklist_add template already includes AC3 duplicate checking
                # Having both creates duplicate functions (addToWorklist AND addToWorklistWithCheck)
                if cap.capability_name == 'duplicate_checker':
                    continue
                # Indent each line of the handler code
                indented_code = '\n'.join('  ' + line if line.strip() else line for line in cap.code.split('\n'))
                handlers_code += f"\n\n{indented_code}"
                fixes.append(f"Generated {cap.capability_name} from config (implements {', '.join(cap.ac_ids)})")

            # Build utilities from rendered capabilities
            # NOTE: Exclude worklist_filter since we handle getFilteredWorklistItems replacement separately
            utilities_code = ""
            for cap in by_injection.get('utilities', []):
                # Skip worklist_filter - it's replaced separately via render_worklist_filter()
                if cap.capability_name == 'worklist_filter':
                    continue
                # Indent each line of the utility code
                indented_code = '\n'.join('  ' + line if line.strip() else line for line in cap.code.split('\n'))
                utilities_code += f"\n\n{indented_code}"
                fixes.append(f"Generated {cap.capability_name} utility from config")

            # Build effects from rendered capabilities
            effects_code = ""
            for cap in by_injection.get('effects', []):
                effects_code += f"\n\n  {cap.code}"
                fixes.append(f"Generated {cap.capability_name} effect from config")

            # IMPORTANT: Only inject hook destructure + formData at the hook location
            # Handlers must be injected AFTER showNotification is defined to avoid reference errors
            new_hook_block = hook_destructure

            # Replace the old destructure pattern with new config-driven hook only
            old_destructure_pattern = r"const \{ createNsiItems, updateNsiItems, deleteNsiItems \} = useNsiItems\(\);"
            if re.search(old_destructure_pattern, content):
                old_literal = "const { createNsiItems, updateNsiItems, deleteNsiItems } = useNsiItems();"
                content = content.replace(old_literal, new_hook_block)
                fixes.append(f"Wired use{entity_pascal} hook with config-driven capabilities")

            # CRITICAL: Replace getFilteredWorklistItems BEFORE injecting handlers
            # Otherwise the injected utilities contain a new getFilteredWorklistItems function
            # and the regex below will match from the new one to the old one, capturing
            # everything in between (including showModal, closeModal, etc.)
            old_getFiltered_pattern = r"const getFilteredWorklistItems = \(\) => \{[\s\S]*?return worklistItems;[\s\S]*?\};"
            new_getFiltered = renderer.render_worklist_filter(entity_name)
            if re.search(old_getFiltered_pattern, content):
                content = re.sub(old_getFiltered_pattern, new_getFiltered, content)
                fixes.append("Updated getFilteredWorklistItems using config-driven worklist_filter template")

            # Inject handlers AFTER showNotification is defined
            # Find the showNotification function and inject handlers after it
            # Pattern matches both JS and TS versions (with or without type annotations)

    def _transform_maintenance_screen_to_dynamic(self, content: str, fixes: list, entity_name: str) -> str:
        """
        Transform the MaintenanceScreen hardcoded sample items into dynamic entity.map().

        CONFIG-DRIVEN: Uses entity_name and database schema for field names.

        The input TSX has:
        - {/* Sample Item Cards */} with hardcoded Item #1, Item #2

        This transforms it to:
        - {/* Dynamic Items from API */}
        - {entity_var.length === 0 ? (... no items message ...) : entity_var.map((item, index) => (... item card ...))}
        """
        import re

        # CONFIG-DRIVEN: Generate entity variable name from entity_name
        entity_var = entity_name.replace('_', '')  # e.g., my_items -> myitems
        entity_label = entity_name.replace('_', ' ').title()  # e.g., my_items -> My Items
        primary_key = self._get_primary_key(entity_name)  # Get from schema
        fields = self._get_entity_fields(entity_name)

        # Determine if entity has special fields (from schema)
        # CONFIG-DRIVEN: Check for special handling flag field from config or common patterns
        field_names = [f.get('name', '') for f in fields]
        special_flag_field = self.app_config.get('special_flag_field')  # From config if specified
        if not special_flag_field:
            # Auto-detect common special flag field patterns
            flag_patterns = ['is_special', 'flagged', 'priority_flag', 'requires_handling', 'special_handling']
            special_flag_field = next((f for f in flag_patterns if f in field_names), None)
        has_special_flag = special_flag_field is not None and special_flag_field in field_names
        has_status = 'status' in field_names
        has_description = 'description' in field_names
        has_quantity = 'quantity' in field_names
        has_disposition = 'disposition' in field_names

        # Pattern to match the entire Sample Item Cards section
        # From "{/* Sample Item Cards */}" to the closing of the wrapper div before Action Buttons
        sample_items_pattern = r'\{/\* Sample Item Cards \*/\}\s*<div style=\{\{ display: \'flex\', flexDirection: \'column\', gap: \'30px\' \}\}>[\s\S]*?</div>\s*</div>\s*(?=\s*\{/\* Action Buttons \*/\})'

        # CONFIG-DRIVEN: Build dynamic JSX using entity config
        # All variable names and field references come from config, not hardcoded
        dynamic_items = self._build_entity_list_jsx(
            entity_var=entity_var,
            entity_label=entity_label,
            primary_key=primary_key,
            has_special_flag=has_special_flag,
            special_flag_field=special_flag_field,
            has_status=has_status,
            has_description=has_description,
            has_quantity=has_quantity,
            has_disposition=has_disposition,
            fields=fields
        )

        if re.search(sample_items_pattern, content):
            content = re.sub(sample_items_pattern, dynamic_items, content)
            fixes.append(f"Transformed MaintenanceScreen to use dynamic {entity_var}.map() from API (config-driven)")
        else:
            logger.warning("Could not find Sample Item Cards pattern in MaintenanceScreen")

        return content

    def _cleanup_orphan_components(self) -> Dict[str, Any]:
        """
        Clean up orphan components that are not used and don't have valid hooks.
        Note: Modals are kept as they are needed for user stories (US3, US4, US5).
        Modal hooks will be generated by step_7d.
        """
        try:
            removed = []
            src_dir = self.output_path.parent  # e.g., Generated/frontend/src
            components_dir = src_dir / 'components'

            # Read App.tsx to find what's actually imported
            with open(self.output_path, 'r') as f:
                app_content = f.read()

            # Keep modals - they are needed for user stories and hooks will be generated by step_7d
            # Only clean up non-modal orphan components
            if components_dir.exists():
                for component_file in components_dir.glob('*.tsx'):
                    component_name = component_file.stem
                    # Skip index files and modal-related components
                    if component_name == 'index':

                    if component_name in app_content:

                    # Don't remove - just log for now

    def _generate_api_service(self) -> str:
        """Generate the services/api.ts file for API calls."""
        return '''/**
 * API Client Service
 * Provides a centralized HTTP client for API calls
 */

const API_BASE_URL = process.env.REACT_APP_API_URL || '/api/v1';

interface ApiResponse<T = any> {
  data: T;
  status: number;
  ok: boolean;
}

class ApiClient {
  private baseUrl: string;

  constructor(baseUrl: string = API_BASE_URL) {
    this.baseUrl = baseUrl;
  }

  private async request<T>(
    endpoint: string,
    options: RequestInit = {}
  ): Promise<ApiResponse<T>> {
    const url = `${this.baseUrl}${endpoint}`;

    const defaultHeaders: HeadersInit = {
      'Content-Type': 'application/json',
    };

    const config: RequestInit = {
      ...options,
      headers: {
        ...defaultHeaders,
        ...options.headers,
      },
    };

    try {
      const response = await fetch(url, config);
      const data = response.ok ? await response.json() : null;

      return {
        data,
        status: response.status,
        ok: response.ok,
      };
    } catch (error) {
      console.error('API request failed:', error);
      throw error;
    }
  }

  async get<T>(endpoint: string): Promise<ApiResponse<T>> {
    return this.request<T>(endpoint, { method: 'GET' });
  }

  async post<T>(endpoint: string, data: any): Promise<ApiResponse<T>> {
    return this.request<T>(endpoint, {
      method: 'POST',
      body: JSON.stringify(data),
    });
  }

  async put<T>(endpoint: string, data: any): Promise<ApiResponse<T>> {
    return this.request<T>(endpoint, {
      method: 'PUT',
      body: JSON.stringify(data),
    });
  }

  async patch<T>(endpoint: string, data: any): Promise<ApiResponse<T>> {
    return this.request<T>(endpoint, {
      method: 'PATCH',
      body: JSON.stringify(data),
    });
  }

  async delete<T>(endpoint: string): Promise<ApiResponse<T>> {
    return this.request<T>(endpoint, { method: 'DELETE' });
  }
}

export const apiClient = new ApiClient();
export default apiClient;
'''

    def _generate_entity_hook(self, entity_name: str) -> str:

        entity_plural = entity_lower if entity_lower.endswith('s') else entity_lower + 's'
